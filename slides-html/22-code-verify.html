static boolean isValidSignature(Map<String, String> params) {    String received = params.get("vnp_SecureHash");    if (received == null || received.isBlank()) return false;     Map<String, String> clone = new TreeMap<>(params);    clone.remove("vnp_SecureHash");    clone.remove("vnp_SecureHashType");     return hmacSHA512(HASH_SECRET, buildQueryString(clone)).equalsIgnoreCase(received);}
